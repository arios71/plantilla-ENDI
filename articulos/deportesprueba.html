    <!-- Script de Autoguardado, Contador, Alertas y Funciones de Exportación -->

    <script>

        const editor = document.getElementById('editor');

        const wordCountDisplay = document.getElementById('word-count');

        const alertMsg = document.getElementById('alert-msg');

        const globalFilenameInput = document.getElementById('globalFilename');

        

        function actualizarContador() {

            const texto = editor.innerText.trim();

            const palabras = texto === "" ? 0 : texto.split(/\s+/).length;

            

            wordCountDisplay.textContent = `Palabras: ${palabras}`;


            // Limpiar clases de alertas previas

            alertMsg.className = "word-alert";


            // Gestión de los 3 umbrales actualizados

            if (palabras >= 1200) {

                alertMsg.textContent = "¡Tope crítico (1,200 palabras)!";

                alertMsg.classList.add("alert-red");

                alertMsg.style.display = "inline-block";

            } else if (palabras >= 1100) {

                alertMsg.textContent = "¡Límite segunda página (1,100 palabras)!";

                alertMsg.classList.add("alert-orange");

                alertMsg.style.display = "inline-block";

            } else if (palabras >= 650) {

                alertMsg.textContent = "650 palabras (primera página)";

                alertMsg.classList.add("alert-yellow");

                alertMsg.style.display = "inline-block";

            } else {

                alertMsg.style.display = "none";

            }

        }


        // Recuperar borrador y nombre de archivo guardados al cargar

        if (localStorage.getItem('articulo_fijo_borrador')) {

            editor.innerHTML = localStorage.getItem('articulo_fijo_borrador');

            actualizarContador();

        }

        if (localStorage.getItem('articulo_fijo_nombre')) {

            globalFilenameInput.value = localStorage.getItem('articulo_fijo_nombre');

        }


        // Eventos de autoguardado en tiempo real

        editor.addEventListener('input', function() {

            actualizarContador();

            localStorage.setItem('articulo_fijo_borrador', editor.innerHTML);

        });


        globalFilenameInput.addEventListener('input', function() {

            localStorage.setItem('articulo_fijo_nombre', globalFilenameInput.value);

        });


        // 1. Guardar como Word (.doc)

        function exportarWord() {

            const tituloSeccion = "DEPORTES - El Nuevo Día";

            const autor = "Antolín Maldonado Ríos";

            const contenidoTexto = editor.innerHTML;


            const contenidoCompleto = `

                <html xmlns:o='urn:schemas-microsoft-com:office:office' xmlns:w='urn:schemas-microsoft-com:office:word' xmlns='http://www.w3.org/TR/REC-html40'>

                <head><meta charset='utf-8'><title>Artículo Deportes</title></head>

                <body style="font-family: 'Times New Roman', Times, serif; font-size: 11pt; line-height: 1.35;">

                    <h1 style="color: #c0392b; font-family: Arial, sans-serif;">${tituloSeccion}</h1>

                    <p style="font-family: Arial, sans-serif; font-size: 10pt; color: #555;"><b>Autor:</b> ${autor}</p>

                    <hr style="border: none; border-top: 1px solid #ccc; margin: 15px 0;"/>

                    ${contenidoTexto}

                </body>

                </html>

            `;


            const blob = new Blob(['\ufeff' + contenidoCompleto], { type: 'application/msword' });

            const url = URL.createObjectURL(blob);

            const downloadLink = document.createElement("a");

            downloadLink.href = url;

            downloadLink.download = (globalFilenameInput.value.trim() || "Articulo_Deportes") + ".doc";

            document.body.appendChild(downloadLink);

            downloadLink.click();

            document.body.removeChild(downloadLink);

        }

            // Cargar token guardado en la sesión previa si existe

            const tokenGuardado = sessionStorage.getItem('github_pat');

            if (tokenGuardado) {

                document.getElementById('ghToken').value = tokenGuardado;

            }

        }


        function cerrarModalGitHub() {

            document.getElementById('githubModal').style.display = 'none';

        }


        async function ejecutarPublicacionGitHub() {

            const filename = document.getElementById('ghFilename').value.trim();

            const token = document.getElementById('ghToken').value.trim();


            if (!filename || !token) {

                alert("Por favor, completa el nombre del archivo y el token de GitHub.");

                return;

            }


            // Guardar token temporalmente en la sesión

            sessionStorage.setItem('github_pat', token);


            // Datos de tu repositorio

            const repoOwner = "arios71";

            const repoName = "plantilla-ENDI";

            const ruta = `articulos/${filename}`; // Se guardará dentro de una carpeta /articulos/

            const url = `https://api.github.com/repos/${repoOwner}/${repoName}/contents/${ruta}`;


            // Armar una página HTML estructurada para el artículo publicado

            const htmlCompletoArticulo = `<!DOCTYPE html>

<html lang="es">

<head>

    <meta charset="UTF-8">

    <title>Deportes - El Nuevo Día</title>

    <style>

        body { font-family: "Times New Roman", Times, serif; font-size: 11pt; line-height: 1.35; max-width: 700px; margin: 30px auto; padding: 20px; color: #111; }

        h1 { color: #c0392b; font-family: Arial, sans-serif; }

        .byline { font-family: Arial, sans-serif; font-size: 9pt; color: #555; margin-bottom: 20px; }

        p { margin: 0 0 10px 0; text-align: justify; text-indent: 1.5em; }

        p:first-child { text-indent: 0; }

    </style>

</head>

<body>

    <h1>DEPORTES</h1>

    <div class="byline"><b>Por:</b> Antolín Maldonado Ríos | El Nuevo Día</div>

    <hr style="border: none; border-top: 1px solid #ccc; margin: 15px 0;"/>

    ${editor.innerHTML}

</body>

</html>`;


            // Codificar contenido en Base64 soportando tildes y caracteres especiales

            const contenidoBase64 = btoa(unescape(encodeURIComponent(htmlCompletoArticulo)));


            const bodyData = {

                message: `Publicando nuevo artículo desde la web: ${filename}`,

                content: contenidoBase64,

                branch: "main"

            };


            // Verificar si el archivo ya existe para obtener su SHA (necesario para actualizarlo si se repite el nombre)

            let sha = null;

            try {

                const respGet = await fetch(url, {

                    headers: { 'Authorization': `Bearer ${token}`, 'Accept': 'application/vnd.github+json' }

                });

                if (respGet.ok) {

                    const dataFile = await respGet.json();

                    sha = dataFile.sha;

                }

            } catch (e) {

                // Si no existe, no pasa nada, se creará nuevo

            }


            if (sha) {

                bodyData.sha = sha; // GitHub requiere el SHA si el archivo ya existe

            }


            try {

                const respuesta = await fetch(url, {

                    method: 'PUT',

                    headers: {

                        'Authorization': `Bearer ${token}`,

                        'Content-Type': 'application/json',

                        'Accept': 'application/vnd.github+json'

                    },

                    body: JSON.stringify(bodyData)

                });


                if (!respuesta.ok) {

                    const errorData = await respuesta.json();

                    throw new Error(errorData.message || 'Error desconocido en la API');

                }


                alert("¡Artículo publicado correctamente en GitHub Pages!");

                cerrarModalGitHub();

            } catch (error) {

                console.error(error);

                alert("No se pudo publicar el artículo. Error: " + error.message);

            }

        }

    </script>

</body>

</html>
